cmake_minimum_required(VERSION 3.20)
project(CodexCrack LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(src/cpp/a)
set(CV440 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/opencv440")
set(FROZEN "${CMAKE_CURRENT_SOURCE_DIR}/baseline/frozen_a/src")
if(NOT EXISTS "${CV440}/lib/opencv_world440.lib")
  message(FATAL_ERROR "Missing project-local OpenCV4.4.0 SDK")
endif()

# A-prime-3 isolated seed-gate observer. Not a production replacement.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/research/aprime3/src/CrackV4_ap3_observer.cpp")
  add_library(CrackV4AP3Observer SHARED research/aprime3/src/CrackV4_ap3_observer.cpp)
  target_include_directories(CrackV4AP3Observer PRIVATE "${FROZEN}")
  target_compile_definitions(CrackV4AP3Observer PRIVATE CRACK_V4_EXPORTS NOMINMAX)
  target_link_libraries(CrackV4AP3Observer PRIVATE opencv440)
  target_compile_options(CrackV4AP3Observer PRIVATE /O2 /fp:precise /utf-8)
endif()
add_library(opencv440 SHARED IMPORTED)
set_target_properties(opencv440 PROPERTIES IMPORTED_IMPLIB "${CV440}/lib/opencv_world440.lib" IMPORTED_LOCATION "${CV440}/bin/opencv_world440.dll" INTERFACE_INCLUDE_DIRECTORIES "${CV440}/include")

# Candidate only. This target keeps Frozen A and the production A target
# untouched while testing anchor/endpoint decoupling.
set(DECOUPLED_CANDIDATE "${CMAKE_CURRENT_SOURCE_DIR}/research/decoupled_candidate")
add_library(CrackV4DecoupledCandidate SHARED "${DECOUPLED_CANDIDATE}/CrackV4.cpp")
target_include_directories(CrackV4DecoupledCandidate PUBLIC "${DECOUPLED_CANDIDATE}/include")
target_include_directories(CrackV4DecoupledCandidate PRIVATE "${DECOUPLED_CANDIDATE}/internal" "${DECOUPLED_CANDIDATE}/include")
target_compile_definitions(CrackV4DecoupledCandidate PRIVATE CRACK_V4_EXPORTS NOMINMAX)
target_link_libraries(CrackV4DecoupledCandidate PRIVATE opencv440)
target_compile_options(CrackV4DecoupledCandidate PRIVATE /O2 /fp:precise /utf-8)
add_custom_command(TARGET CrackV4DecoupledCandidate POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CV440}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:CrackV4DecoupledCandidate>")
add_executable(CrackReadOnlyBench src/cpp/read_only_bench.cpp)
target_include_directories(CrackReadOnlyBench PRIVATE "${FROZEN}")
target_link_libraries(CrackReadOnlyBench PRIVATE opencv440)
# Separate build artifact. The frozen DLL is never a build target or overwritten.
add_library(CrackV4Candidate SHARED "${FROZEN}/CrackV4.cpp")
add_library(CrackV4GapResume SHARED research/gap_resume/gap_resume.cpp)
target_compile_definitions(CrackV4GapResume PRIVATE CRACK_V4_EXPORTS NOMINMAX)
target_link_libraries(CrackV4GapResume PRIVATE opencv440)
target_compile_options(CrackV4GapResume PRIVATE /O2 /fp:precise /utf-8)
add_library(CrackGateProbe SHARED research/gate_adapt/probe.cpp)
target_compile_definitions(CrackGateProbe PRIVATE CRACK_V4_EXPORTS NOMINMAX)
target_link_libraries(CrackGateProbe PRIVATE opencv440)
target_compile_options(CrackGateProbe PRIVATE /O2 /fp:precise /utf-8)
target_include_directories(CrackV4Candidate PUBLIC "${FROZEN}")
target_compile_definitions(CrackV4Candidate PRIVATE CRACK_V4_EXPORTS NOMINMAX)
target_link_libraries(CrackV4Candidate PRIVATE opencv440)
foreach(t CrackReadOnlyBench CrackV4Candidate)
  if(MSVC)
    target_compile_options(${t} PRIVATE /O2 /fp:precise /utf-8)
  endif()
  add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CV440}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:${t}>")
endforeach()

# A-prime-2 observer clone: isolated output, never installed over production.
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/research/aprime2/src/native_shadow/CrackV4_ap2.cpp")
  add_library(CrackV4AP2Shadow SHARED research/aprime2/src/native_shadow/CrackV4_ap2.cpp)
  target_include_directories(CrackV4AP2Shadow PRIVATE "${FROZEN}")
  target_compile_definitions(CrackV4AP2Shadow PRIVATE CRACK_V4_EXPORTS NOMINMAX)
  target_link_libraries(CrackV4AP2Shadow PRIVATE opencv440)
  target_compile_options(CrackV4AP2Shadow PRIVATE /O2 /fp:precise /utf-8)
  add_executable(AP2DualBench research/aprime2/src/native_shadow/dual_bench.cpp)
  target_include_directories(AP2DualBench PRIVATE "${FROZEN}")
  target_link_libraries(AP2DualBench PRIVATE opencv440)
  target_compile_options(AP2DualBench PRIVATE /O2 /fp:precise /utf-8)
endif()

add_subdirectory(src/cpp/v510)

add_subdirectory(src/cpp/v512)

add_subdirectory(src/cpp/v63)
