cmake_minimum_required(VERSION 3.20)
project(CrackV4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CRACK_V4_OPENCV_VERSION "4.4.0" CACHE STRING "Exact OpenCV version used by the detector")
if(NOT OpenCV_DIR AND WIN32 AND EXISTS "D:/opencv-4.4.0/build/OpenCVConfig.cmake")
 set(OpenCV_DIR "D:/opencv-4.4.0/build" CACHE PATH "OpenCV SDK build directory")
endif()
find_package(OpenCV ${CRACK_V4_OPENCV_VERSION} EXACT REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(CrackV4 SHARED CrackV4.cpp)
option(CRACK_V4_PROFILE "Enable internal stage profiling for benchmarks" OFF)
if(CRACK_V4_PROFILE)
 target_compile_definitions(CrackV4 PRIVATE CRACK_V4_PROFILE)
endif()
target_compile_definitions(CrackV4 PRIVATE CRACK_V4_EXPORTS NOMINMAX)
target_include_directories(CrackV4 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
target_link_libraries(CrackV4 PRIVATE ${OpenCV_LIBS})
if(MSVC)
 target_compile_options(CrackV4 PRIVATE /O2 /fp:precise /utf-8 /W4)
endif()
add_executable(CrackV4Example example.cpp)
target_include_directories(CrackV4Example PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(CrackV4Example PRIVATE CrackV4 ${OpenCV_LIBS})
if(MSVC)
 target_compile_options(CrackV4Example PRIVATE /utf-8)
endif()
add_executable(CrackV4V2Example example_v2.cpp)
target_include_directories(CrackV4V2Example PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(CrackV4V2Example PRIVATE CrackV4 ${OpenCV_LIBS})
if(MSVC)
 target_compile_options(CrackV4V2Example PRIVATE /utf-8)
endif()
add_executable(CrackV4ExperimentExample example_experiments.cpp)
target_include_directories(CrackV4ExperimentExample PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(CrackV4ExperimentExample PRIVATE CrackV4 ${OpenCV_LIBS})
if(MSVC)
 target_compile_options(CrackV4ExperimentExample PRIVATE /utf-8)
endif()
