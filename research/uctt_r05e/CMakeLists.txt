cmake_minimum_required(VERSION 3.20)
project(UCTTR05E LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CV "${ROOT}/third_party/opencv440")
add_library(B0 SHARED "${ROOT}/src/cpp/a/CrackV4.cpp")
add_executable(R03Original "${ROOT}/research/uctt_r03/contracts.cpp")
add_executable(EvidenceContracts contracts.cpp)
target_compile_definitions(EvidenceContracts PRIVATE R05E_BUDGET)
set(TARGETS B0 R03Original EvidenceContracts)
if(EXISTS "${ROOT}/output/build/uctt_r05e_source/CrackV4.cpp")
 foreach(t B1 B2)
  add_library(${t} SHARED "${ROOT}/output/build/uctt_r05e_source/CrackV4.cpp")
  list(APPEND TARGETS ${t})
 endforeach()
 target_compile_definitions(B2 PRIVATE R05E_BUDGET)
 add_library(EVIDENCE_DP_PROBE SHARED probe.cpp)
 list(APPEND TARGETS EVIDENCE_DP_PROBE)
endif()
foreach(t ${TARGETS})
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CV}/include" "${ROOT}/src/cpp/a/include" "${ROOT}/src/cpp/a/internal")
 target_compile_definitions(${t} PRIVATE CRACK_V4_EXPORTS NOMINMAX)
 target_compile_options(${t} PRIVATE /O2 /fp:precise /utf-8)
 target_link_libraries(${t} PRIVATE "${CV}/lib/opencv_world440.lib")
 add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CV}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:${t}>")
endforeach()
