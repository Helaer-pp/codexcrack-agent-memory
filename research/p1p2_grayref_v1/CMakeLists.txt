cmake_minimum_required(VERSION 3.20)
project(P1P2GrayRefV1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CV "${ROOT}/third_party/opencv440")
add_library(P1P2GrayRefV1 SHARED runtime.cpp)
target_include_directories(P1P2GrayRefV1 PRIVATE "${CV}/include" "${ROOT}/src/cpp/a/include")
target_compile_definitions(P1P2GrayRefV1 PRIVATE NOMINMAX)
target_compile_options(P1P2GrayRefV1 PRIVATE /O2 /fp:strict /utf-8)
target_link_libraries(P1P2GrayRefV1 PRIVATE "${CV}/lib/opencv_world440.lib")
add_custom_command(TARGET P1P2GrayRefV1 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CV}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:P1P2GrayRefV1>")
