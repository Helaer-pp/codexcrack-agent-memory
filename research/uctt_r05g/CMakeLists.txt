cmake_minimum_required(VERSION 3.20)
project(UCTTR05G LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CV "${ROOT}/third_party/opencv440")
add_library(R05G SHARED runtime.cpp)
option(R05G_ENABLE_BEAM "Enable controlled G2 variants after G1 evidence" OFF)
if(R05G_ENABLE_BEAM)
 target_compile_definitions(R05G PRIVATE R05G_ENABLE_BEAM)
endif()
add_executable(R03Original "${ROOT}/research/uctt_r03/contracts.cpp")
foreach(t R05G R03Original)
 target_include_directories(${t} PRIVATE "${CV}/include" "${ROOT}/src/cpp/a/include")
 target_compile_definitions(${t} PRIVATE NOMINMAX CRACK_V4_EXPORTS)
 target_compile_options(${t} PRIVATE /O2 /fp:precise /utf-8)
 target_link_libraries(${t} PRIVATE "${CV}/lib/opencv_world440.lib")
 add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CV}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:${t}>")
endforeach()
