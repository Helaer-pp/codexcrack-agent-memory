cmake_minimum_required(VERSION 3.20)
project(GrayRefV2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
while(NOT EXISTS "${ROOT}/config/project.json")
 get_filename_component(PARENT "${ROOT}" DIRECTORY)
 if(PARENT STREQUAL ROOT)
  message(FATAL_ERROR "Cannot find active project")
 endif()
 set(ROOT "${PARENT}")
endwhile()
set(CV "${ROOT}/third_party/opencv440")
add_library(GrayRefV2 SHARED runtime.cpp)
target_include_directories(GrayRefV2 PRIVATE "${CV}/include")
target_compile_definitions(GrayRefV2 PRIVATE NOMINMAX)
target_compile_options(GrayRefV2 PRIVATE /O2 /fp:strict /utf-8)
target_link_libraries(GrayRefV2 PRIVATE "${CV}/lib/opencv_world440.lib")
add_custom_command(TARGET GrayRefV2 POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CV}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:GrayRefV2>")
