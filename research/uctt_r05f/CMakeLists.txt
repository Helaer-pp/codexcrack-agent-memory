cmake_minimum_required(VERSION 3.20)
project(UCTTR05F LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CV "${ROOT}/third_party/opencv440")
add_library(A0_CLONE SHARED "${ROOT}/output/build/uctt_r05f_source/CrackV4.cpp")
add_executable(R03Original "${ROOT}/research/uctt_r03/contracts.cpp")
set(TARGETS A0_CLONE R03Original)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/runtime.cpp")
 add_library(R05F SHARED runtime.cpp)
 list(APPEND TARGETS R05F)
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/budget_probe.cpp")
 add_library(S1BudgetProbe SHARED budget_probe.cpp)
 set_target_properties(S1BudgetProbe PROPERTIES OUTPUT_NAME R05F RUNTIME_OUTPUT_DIRECTORY "${ROOT}/output/build/uctt_r05f_probe" ARCHIVE_OUTPUT_DIRECTORY "${ROOT}/output/build/uctt_r05f_probe")
 list(APPEND TARGETS S1BudgetProbe)
endif()
foreach(t ${TARGETS})
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CV}/include" "${ROOT}/src/cpp/a/include" "${ROOT}/src/cpp/a/internal")
 target_compile_definitions(${t} PRIVATE CRACK_V4_EXPORTS NOMINMAX)
 target_compile_options(${t} PRIVATE /O2 /fp:precise /utf-8)
 target_link_libraries(${t} PRIVATE "${CV}/lib/opencv_world440.lib")
 add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CV}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:${t}>")
endforeach()
