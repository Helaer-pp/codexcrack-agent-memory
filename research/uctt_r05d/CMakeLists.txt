cmake_minimum_required(VERSION 3.20)
project(UCTTR05D LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CV "${ROOT}/third_party/opencv440")
add_library(A0_SAME_SOURCE SHARED "${ROOT}/src/cpp/a/CrackV4.cpp")
add_library(D0_TRACE_ONLY SHARED "${ROOT}/output/build/uctt_r05d_source/CrackV4.cpp")
add_library(D1_OFFLINE_ONLY SHARED "${ROOT}/output/build/uctt_r05d_source/CrackV4.cpp")
add_library(D2_LOCAL_AUDIT SHARED "${ROOT}/output/build/uctt_r05d_source/CrackV4.cpp")
target_compile_definitions(D1_OFFLINE_ONLY PRIVATE R05D_OFFLINE)
foreach(t A0_SAME_SOURCE D0_TRACE_ONLY D1_OFFLINE_ONLY D2_LOCAL_AUDIT)
 target_include_directories(${t} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CV}/include" "${ROOT}/src/cpp/a/include" "${ROOT}/src/cpp/a/internal")
 target_compile_definitions(${t} PRIVATE CRACK_V4_EXPORTS NOMINMAX)
 target_compile_options(${t} PRIVATE /O2 /fp:precise /utf-8)
 target_link_libraries(${t} PRIVATE "${CV}/lib/opencv_world440.lib")
 add_custom_command(TARGET ${t} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CV}/bin/opencv_world440.dll" "$<TARGET_FILE_DIR:${t}>")
endforeach()
